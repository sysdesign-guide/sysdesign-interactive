  <script>

    const urlParams = new URLSearchParams(window.location.search);

    const promptPath = urlParams.get('prompt') || 'v1/prompts/ch01-tco-simulator.md';


    fetch(promptPath)

      .then(res => res.text())

      .then(text => {

        document.getElementById('status').innerText = 'Промпт успешно загружен:';

        document.getElementById('prompt-content').innerText = text;

        

        // Прямой запуск для Gemini через URL

        const geminiUrl = `https://gemini.google.com/app?prompt=${encodeURIComponent(text)}`;

        document.getElementById('gemini-btn').href = geminiUrl;


        // Копирование в буфер обмена для Copilot/ChatGPT/Claude

        document.getElementById('copy-btn').onclick = () => {

          navigator.clipboard.writeText(text);

          alert('Промпт скопирован в буфер обмена!');

        };

      })

      .catch(err => {

        document.getElementById('status').innerText = 'Ошибка загрузки промпта.';

      });

  </script>
